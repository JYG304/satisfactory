    function syncFloorShapeDefToRealm() {
      const realm = (typeof CURRENT_REALM !== 'undefined') ? CURRENT_REALM : 'qi';
      const spec = (typeof SPEC_DISC !== 'undefined') ? SPEC_DISC : 'combat';
      for (const [k, def] of Object.entries(FLOOR_SHAPE_DEFS)) {
        if (def.variants) {
          let variantKey = realm;
          if (realm === 'qi') {
            variantKey = (spec === 'combat' && def.variants['qi_combat']) ? 'qi_combat' : 'qi';
          }
          const v = def.variants[variantKey] || def.variants[realm] || def.variants['qi'];
          if (v) {
            def.nodes = v.nodes;
            if (v.links) def.links = v.links;
            def.activeVariant = variantKey;
            def.variantName = v.variantName;
            def.variantDesc = v.desc || def.desc;
          }
        }
      }
    }

        shapeDef.nodes.forEach(node => {
          nodeMap[node.id] = node;
          let isAligned = !!(hitSet && hitSet.has(node.id));
          if (!isAligned) {
            for (const [rId, rMeta] of Object.entries(DISK_RINGS)) {
              if (rMeta.tier > (counts[rMeta.discKey] || 1)) continue;
              for (let sIdx = 0; sIdx < rMeta.slotCount; sIdx++) {
                const p = getSlotWorldPos(rId, sIdx);
                if (Math.hypot(p.x - node.x, p.y - node.y) <= 26) {
                  const sKey = `${rId}_${sIdx}`;
                  if (BOARD_SLOTS[sKey] && BOARD_SLOTS[sKey].chip) {
                    isAligned = true;
                    break;
                  }
                }
              }
              if (isAligned) break;
            }
          }
          alignedMap[node.id] = isAligned;
        });
      }

      let linksSvg = '';
      if (shapeDef.links && shapeDef.nodes) {
        shapeDef.links.forEach(([idA, idB]) => {
          const nA = nodeMap[idA];
          const nB = nodeMap[idB];
          if (nA && nB) {
            const isFullActive = alignedMap[idA] && alignedMap[idB];
            const linkColor = (nA.disc === 'combat' && nB.disc === 'combat') 
              ? '#f43f5e' 
              : ((nA.disc === 'industry' && nB.disc === 'industry') ? '#f59e0b' : ((nA.disc === 'support' && nB.disc === 'support') ? '#00f0ff' : '#94a3b8'));
            if (isFullActive) {
              linksSvg += `
                <line x1="${nA.x}" y1="${nA.y}" x2="${nB.x}" y2="${nB.y}"
                      stroke="${linkColor}" stroke-width="2.6" stroke-linecap="round" style="filter: drop-shadow(0 0 5px ${linkColor});" />
              `;
            } else {
              linksSvg += `
                <line x1="${nA.x}" y1="${nA.y}" x2="${nB.x}" y2="${nB.y}"
                      stroke="${linkColor}" stroke-width="1.8" stroke-dasharray="5 3" opacity="0.65" />
              `;
            }
          }
        });
      }

      let nodesSvg = '';
      if (shapeDef.nodes) {
        shapeDef.nodes.forEach(node => {
          const isAligned = alignedMap[node.id];
          const badgeX = node.bx !== undefined ? node.bx : node.x + 22;
          const badgeY = node.by !== undefined ? node.by : node.y - 18;

          nodesSvg += `
            <g id="starNodeGroup_${node.id}" class="transition-opacity duration-300">
              ${renderStarGlyph(node, isAligned)}

              <!-- 悬浮说明文字气泡 -->
              <g transform="translate(${badgeX}, ${badgeY})" opacity="${isAligned ? '1' : '0.85'}">
                <rect x="0" y="0" width="76" height="18" rx="4"
                      fill="${isAligned ? '#ffffff' : '#f8fafc'}"
                      stroke="${isAligned ? node.color : '#cbd5e1'}"
                      stroke-width="${isAligned ? '1.5' : '1'}"
                      filter="drop-shadow(0 1px 2px rgba(0,0,0,0.06))" />
                <text x="38" y="12.5" text-anchor="middle"
                      font-family="'Share Tech Mono', monospace" font-size="9.5" font-weight="bold"
                      fill="${isAligned ? node.color : '#475569'}">
                  ${node.num} ${isAligned ? '已导通' : '待对齐'}
                </text>
              </g>
            </g>
          `;
        });
      }

      container.innerHTML = `
        <svg viewBox="0 0 1640 860" class="w-full h-full pointer-events-none">
          <g id="floorShapeLinksLayer" class="transition-all duration-300">${linksSvg}</g>
          <g id="floorShapeNodesLayer" class="transition-all duration-300">${nodesSvg}</g>
        </svg>
      `;
    }

    function switchFloorShape(shapeKey) {
      if (!FLOOR_SHAPE_DEFS[shapeKey]) return;
      CURRENT_FLOOR_SHAPE = shapeKey;
      syncFloorShapeDefToRealm();
      const activeDef = FLOOR_SHAPE_DEFS[shapeKey];

      // 更新按钮高亮
      ['dipper', 'combat_focus', 'factory_focus'].forEach(k => {
        const btn = document.getElementById('btn_shape_' + k);
        if (btn) {
          if (k === shapeKey) {
            btn.className = 'px-2.5 py-1 rounded bg-slate-900 border border-slate-900 text-white text-[11px] font-bold font-mono-sci shadow-xs';
          } else {
            btn.className = 'px-2.5 py-1 rounded bg-white border border-slate-300 hover:border-slate-800 text-slate-700 text-[11px] font-bold font-mono-sci';
          }
        }
      });

      const badgeEl = document.getElementById('constellationBadge');
      if (badgeEl) badgeEl.textContent = activeDef.name;

      renderFloorBase(true);
      triggerConstellationGuidePreview(2200);
      if (typeof solveStepNetwork === 'function') solveStepNetwork();
    }
    window.switchFloorShape = switchFloorShape;

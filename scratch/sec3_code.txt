    const STAGE_LOGICAL_W = 1640;
    const STAGE_LOGICAL_H = 860;
    let USER_ZOOM_FACTOR = 1.0;
    let STAGE_PAN_X = 0;
    let STAGE_PAN_Y = 0;

    function fitStageToViewport(isPanning = false) {
      const section = document.getElementById('astrolabeStageSection');
      const stage = document.getElementById('diskStage');
      if (!section || !stage) return;
      const availW = section.clientWidth || window.innerWidth;
      const availH = section.clientHeight || (window.innerHeight - 64);
      const baseScale = Math.min((availW - 16) / STAGE_LOGICAL_W, (availH - 16) / STAGE_LOGICAL_H);
      const finalScale = Math.max(0.45, Math.min(3.5, baseScale * USER_ZOOM_FACTOR));
      stage.style.transition = isPanning ? 'none' : 'transform 120ms ease-out';
      stage.style.transform = `translate(${Math.round(STAGE_PAN_X)}px, ${Math.round(STAGE_PAN_Y)}px) scale(${finalScale.toFixed(4)})`;

      const pctEl = document.getElementById('zoomPercentText');
      if (pctEl) pctEl.textContent = `${Math.round(finalScale * 100)}%`;
      const resBadge = document.getElementById('resolutionBadge');
      if (resBadge) {
        const w = window.screen.width * (window.devicePixelRatio || 1);
        resBadge.textContent = w >= 3000 ? '4K超清自适应' : (w >= 2000 ? '2K高清自适应' : '全屏自适应');
      }
    }
    window.fitStageToViewport = fitStageToViewport;

    function adjustUserZoom(delta) {
      USER_ZOOM_FACTOR = Math.max(0.5, Math.min(2.8, Number((USER_ZOOM_FACTOR + delta).toFixed(2))));
      fitStageToViewport();
    }
    window.adjustUserZoom = adjustUserZoom;

    function resetUserZoom() {
      USER_ZOOM_FACTOR = 1.0;
      STAGE_PAN_X = 0;
      STAGE_PAN_Y = 0;
      fitStageToViewport();
    }
    window.resetUserZoom = resetUserZoom;

    function getActiveRingCounts() {

        qi:         { primary: 2, secondary: 1 },

        support:  SPEC_DISC === 'support'  ? cfg.primary : cfg.secondary

    function getOuterDockRadius(discKey) {
      const counts = getActiveRingCounts();
      const maxTier = counts[discKey] || 1;
      const maxBasicR = DISC_CONFIG[discKey].rings[maxTier - 1].radius;
      return maxBasicR + 36;
    }

    const DOCK_BASE_SLOTS = [

    let OUTER_DOCK_SLOTS = {};
    function initOuterDockSlots() {
      const prev = Object.assign({}, OUTER_DOCK_SLOTS);
      OUTER_DOCK_SLOTS = {};
      for (const dKey of ['support', 'industry', 'combat']) {
        DOCK_BASE_SLOTS.forEach(bs => {
          const dockKey = `dock_${dKey}_${bs.idx}`;
          OUTER_DOCK_SLOTS[dockKey] = {
            dockKey,
            discKey: dKey,
            ringId: `${dKey}_dock`,
            dockIndex: bs.idx,
            baseDeg: bs.baseDeg,
            label: bs.label,
            coreChip: prev[dockKey] ? prev[dockKey].coreChip : null
          };
        });
      }
    }

    // 计算独立外圈插槽旋转后的实时世界坐标与当前角度
    function getOuterDockWorldPos(dockKey) {
      const dSlot = OUTER_DOCK_SLOTS[dockKey];
      if (!dSlot) return { x: 0, y: 0, angleDeg: 0, radius: 100 };
      const dMeta = DISC_CONFIG[dSlot.discKey];
      const dockRing = DISK_RINGS[dSlot.ringId];
      const r = getOuterDockRadius(dSlot.discKey);
      if (dockRing) dockRing.radius = r;
      const rot = dockRing ? dockRing.currentAngle : 0;
      const effDeg = (dSlot.baseDeg + rot + 720) % 360;
      const rad = (effDeg * Math.PI) / 180;
      return {
        x: dMeta.cx + r * Math.cos(rad),
        y: dMeta.cy + r * Math.sin(rad),
        angleDeg: effDeg,
        radius: r,
        cx: dMeta.cx,
        cy: dMeta.cy
      };
    }

    let BOARD_SLOTS = {};
    function initBoardSlots() {
      const prev = Object.assign({}, BOARD_SLOTS);
      BOARD_SLOTS = {};
      for (const [ringKey, ringMeta] of Object.entries(DISK_RINGS)) {
        if (ringMeta.isDockRing) continue;
        for (let i = 0; i < ringMeta.slotCount; i++) {
          const sKey = `${ringKey}_${i}`;
          BOARD_SLOTS[sKey] = {
            ringId: ringKey,
            discKey: ringMeta.discKey,
            tier: ringMeta.tier,
            slotIndex: i,
            chip: prev[sKey] ? prev[sKey].chip : null
          };
        }
      }
      initOuterDockSlots();
    }

    function renderDynamicSvgDiscCasings() {
      const g = document.getElementById('dynamicSvgDiscCasings');
      if (!g) return;
      const counts = getActiveRingCounts();
      let svgHtml = '';

      for (const [dKey, dMeta] of Object.entries(DISC_CONFIG)) {
        const dockR = getOuterDockRadius(dKey);
        const bezelR = dockR + 18;
        const caliperR = dockR + 30;
        const caliperOuterR = dockR + 42;
        const cx = dMeta.cx;
        const cy = dMeta.cy;

        const a1 = 145 * Math.PI / 180;
        const a2 = 225 * Math.PI / 180;
        const c1x = (cx + caliperR * Math.cos(a1)).toFixed(1);
        const c1y = (cy + caliperR * Math.sin(a1)).toFixed(1);
        const c2x = (cx + caliperR * Math.cos(a2)).toFixed(1);
        const c2y = (cy + caliperR * Math.sin(a2)).toFixed(1);

        const o1x = (cx + caliperOuterR * Math.cos(a1)).toFixed(1);
        const o1y = (cy + caliperOuterR * Math.sin(a1)).toFixed(1);
        const o2x = (cx + caliperOuterR * Math.cos(a2)).toFixed(1);
        const o2y = (cy + caliperOuterR * Math.sin(a2)).toFixed(1);

        let ticksHtml = '';
        for (let deg = 145; deg <= 225; deg += 8) {
          const rad = deg * Math.PI / 180;
          const ix = (cx + caliperR * Math.cos(rad)).toFixed(1);
          const iy = (cy + caliperR * Math.sin(rad)).toFixed(1);
          const ox = (cx + caliperOuterR * Math.cos(rad)).toFixed(1);
          const oy = (cy + caliperOuterR * Math.sin(rad)).toFixed(1);
          const sw = (deg % 16 === 1) ? '1.8' : '1.1';
          ticksHtml += `<line x1="${ix}" y1="${iy}" x2="${ox}" y2="${oy}" stroke="#0f172a" stroke-width="${sw}" />`;

        const hatchH = 44;

        const hatchY = Math.round(cy - bezelR - 20);

        svgHtml += `
          <g id="svgDiscCasing_${dKey}">
            <!-- 45° 精密剖面斜纹工程底框 -->
            <rect x="${hatchX}" y="${hatchY}" width="${hatchW}" height="${hatchH}" fill="url(#blueprintHatch)" stroke="#0f172a" stroke-width="1.1" stroke-dasharray="3 3" opacity="0.85" />

            <path d="M ${c1x} ${c1y} A ${caliperR} ${caliperR} 0 0 1 ${c2x} ${c2y}" fill="none" stroke="#0f172a" stroke-width="1.8" />
            <path d="M ${o1x} ${o1y} A ${caliperOuterR} ${caliperOuterR} 0 0 1 ${o2x} ${o2y}" fill="none" stroke="#cbd5e1" stroke-width="0.8" stroke-dasharray="2 2" />

    // 渲染三盘【可旋转独立外圈】+【内部基础环 (1~4环)】DOM
    function renderDynamicRingsDOM() {
      const container = document.getElementById('dynamicRingsDomContainer');
      if (!container) return;
      const counts = getActiveRingCounts();
      let htmlStr = '';

      for (const [dKey, dMeta] of Object.entries(DISC_CONFIG)) {
        const maxTier = counts[dKey];
        const dockR = getOuterDockRadius(dKey);
        const dockRingObj = DISK_RINGS[`${dKey}_dock`];
        if (dockRingObj) dockRingObj.radius = dockR;

        const badgeEl = document.getElementById('badge_disc_' + dKey);
        if (badgeEl) {
          const isSpec = (SPEC_DISC === dKey);
          badgeEl.textContent = `基础${maxTier}环+独立外圈 · ${isSpec ? '大盘' : '小盘'}`;
          badgeEl.className = isSpec
            ? 'text-[9.5px] font-mono-sci font-bold px-1.5 rounded bg-slate-900 text-amber-300'
            : 'text-[9.5px] font-mono-sci font-bold px-1.5 rounded bg-slate-100 text-slate-600';
        }

        // 1. ★ 渲染最外层【可旋转独立外圈】DOM 轨道（支持直接拖拽/滚轮旋转）
        if (dockRingObj) {
          const dSize = dockR * 2;
          const dLeft = dMeta.cx - dockR;
          const dTop = dMeta.cy - dockR;
          htmlStr += `
            <div id="${dockRingObj.elId}" class="absolute rounded-full border-2 border-dashed border-sky-600/85 ring-trans pointer-events-none"
                 style="width: ${dSize}px; height: ${dSize}px; left: ${dLeft}px; top: ${dTop}px; transform: rotate(${dockRingObj.currentAngle}deg);">
              <div class="absolute inset-2 rounded-full border border-slate-300/80"></div>
              <div class="absolute -top-2.5 left-1/2 -translate-x-1/2 text-[7.5px] font-mono-sci text-sky-900 font-bold bg-sky-50 px-1.5 rounded border border-sky-400 shadow-xs whitespace-nowrap">
                ◎ 独立外圈(可旋转·放中心核)
              </div>
            </div>
          `;
        }

        // 2. 渲染内部基础环 (1~maxTier)
        dMeta.rings.forEach(r => {
          if (r.tier > maxTier) return;
          const ringObj = DISK_RINGS[r.id];
          const isOuterBasic = (r.tier === maxTier);
          const size = r.radius * 2;
          const left = dMeta.cx - r.radius;
          const top = dMeta.cy - r.radius;
          const isSelectedRing = (ACTIVE_INTERACT_RING === r.id);
          const borderClass = isSelectedRing
            ? 'border-2 border-sky-500 shadow-[0_0_12px_rgba(14,165,233,0.35)]'
            : (isOuterBasic ? 'border-2 border-slate-900' : 'border border-slate-600');

            ? 'bg-sky-600 text-white border-sky-400'
            : 'bg-white text-slate-800 border-slate-300';
          htmlStr += `
            <div id="${ringObj.elId}" class="absolute rounded-full ${borderClass} ring-trans pointer-events-none"

    function placeCoreInOuterDock(dockKey, coreChipId) {
      const dSlot = OUTER_DOCK_SLOTS[dockKey];
      const spec = CHIP_SPECS[coreChipId];
      if (!dSlot || !spec || !spec.isCenterCore) return;

      const srcDisc = spec.sourceDisc;
      if (dSlot.discKey === srcDisc) {
        alert(`【${spec.name}】是从${DISC_CONFIG[srcDisc].name}取下的中心核，请将它拖到【其他盘子的独立外圈插槽】上！`);
        return;
      }

      const cState = CENTER_CORE_STATE[srcDisc];
      if (cState.placedAtDock && OUTER_DOCK_SLOTS[cState.placedAtDock]) {
        OUTER_DOCK_SLOTS[cState.placedAtDock].coreChip = null;
      }
      if (dSlot.coreChip && CHIP_SPECS[dSlot.coreChip] && dSlot.coreChip !== coreChipId) {
        const oldSrc = CHIP_SPECS[dSlot.coreChip].sourceDisc;
        CENTER_CORE_STATE[oldSrc].placedAtDock = null;
        CENTER_CORE_STATE[oldSrc].detached = false;
      }

      dSlot.coreChip = coreChipId;
      cState.placedAtDock = dockKey;
      cState.detached = true;
      ACTIVE_SELECTED_CHIP = null;
      updateSelectedChipIndicator();
      if (audio) audio.playConnect();
      refreshDiscRingsAndSlots();
    }
    window.placeCoreInOuterDock = placeCoreInOuterDock;

    function removeCoreFromDock(dockKey, e) {
      if (e) e.stopPropagation();
      if (Date.now() - LAST_CORE_DRAG_TIMESTAMP < 250) return;
      const dSlot = OUTER_DOCK_SLOTS[dockKey];
      if (!dSlot || !dSlot.coreChip) return;
      const spec = CHIP_SPECS[dSlot.coreChip];
      if (spec && spec.isCenterCore) {
        const srcDisc = spec.sourceDisc;
        // ★ 取下来之后直接自动归位到所属盘中心，不在外面弹！
        CENTER_CORE_STATE[srcDisc].placedAtDock = null;
        CENTER_CORE_STATE[srcDisc].detached = false;
        if (ACTIVE_SELECTED_CHIP === dSlot.coreChip) {
          ACTIVE_SELECTED_CHIP = null;
          updateSelectedChipIndicator();
        }
      }
      dSlot.coreChip = null;
      if (audio) audio.playRemove();
      refreshDiscRingsAndSlots();
    }
    window.removeCoreFromDock = removeCoreFromDock;

    function handleDockSlotClick(dockKey, e) {
      if (e) e.stopPropagation();
      if (ACTIVE_SELECTED_CHIP && CHIP_SPECS[ACTIVE_SELECTED_CHIP] && CHIP_SPECS[ACTIVE_SELECTED_CHIP].isCenterCore) {
        placeCoreInOuterDock(dockKey, ACTIVE_SELECTED_CHIP);
      }
    }
    window.handleDockSlotClick = handleDockSlotClick;

    function handleDockSlotDrop(dockKey, e) {
      if (e) {
        e.preventDefault();
        e.stopPropagation();
      }
      const typeId = e.dataTransfer ? e.dataTransfer.getData('text/plain') : null;
      if (typeId && CHIP_SPECS[typeId] && CHIP_SPECS[typeId].isCenterCore) {
        placeCoreInOuterDock(dockKey, typeId);
      }
    }
    window.handleDockSlotDrop = handleDockSlotDrop;

    function handlePlacedDockCoreMouseDown(dockKey, srcDisc, e) {
      if (e.button !== 0) return;
      e.stopPropagation();
      const startX = e.clientX;
      const startY = e.clientY;
      const onMove = (me) => {
        if (Math.hypot(me.clientX - startX, me.clientY - startY) > 5) {
          window.removeEventListener('mousemove', onMove);
          window.removeEventListener('mouseup', onUp);
          if (OUTER_DOCK_SLOTS[dockKey]) OUTER_DOCK_SLOTS[dockKey].coreChip = null;
          CENTER_CORE_STATE[srcDisc].placedAtDock = null;
          CENTER_CORE_STATE[srcDisc].detached = true;
          refreshDiscRingsAndSlots();
          startCorePointerDrag(srcDisc, me);
        }
      };
      const onUp = () => {
        window.removeEventListener('mousemove', onMove);
        window.removeEventListener('mouseup', onUp);
      };
      window.addEventListener('mousemove', onMove);
      window.addEventListener('mouseup', onUp);
    }
    window.handlePlacedDockCoreMouseDown = handlePlacedDockCoreMouseDown;

    /* =========================================================================
       ★ 核心交互：点一下盘心取下中心节点 ➔ 按住拖拽到另外盘子的【可旋转独立外圈插槽】连上导入线！
       ========================================================================= */
    let IS_DRAGGING_CORE = false;
    let DRAGGING_CORE_DISC = null;
    let LAST_CORE_DRAG_TIMESTAMP = 0;

    function handleCenterCoreClick(discKey, e) {
      if (e) e.stopPropagation();
      if (Date.now() - LAST_CORE_DRAG_TIMESTAMP < 250) return;

      const state = CENTER_CORE_STATE[discKey];
      const coreChipId = 'core_' + discKey;

      // 如果当前连在其他盘外圈，点一下直接取下并自动归位回本盘中心！
      if (state.placedAtDock && OUTER_DOCK_SLOTS[state.placedAtDock]) {
        OUTER_DOCK_SLOTS[state.placedAtDock].coreChip = null;
        state.placedAtDock = null;
        state.detached = false;
        if (ACTIVE_SELECTED_CHIP === coreChipId) ACTIVE_SELECTED_CHIP = null;
        updateSelectedChipIndicator();
        if (audio) audio.playRemove();
        refreshDiscRingsAndSlots();
        return;
      }

      if (!state.charged) {
        alert(`【${DISC_CONFIG[discKey].name}中心尚未充能】\n请先让光路射入本盘中心（在基础1环放置「🔺 变轨三角」折向中心），中心激活后直接从中心拖拽到其他盘的【独立外圈插槽】即可！`);
        return;
      }

      // 在中心原位切换选中状态（不弹到外面）
      if (ACTIVE_SELECTED_CHIP !== coreChipId) {
        state.detached = false;
        ACTIVE_SELECTED_CHIP = coreChipId;
        if (audio) audio.playSelect();
      } else {
        state.detached = false;
        ACTIVE_SELECTED_CHIP = null;
        if (audio) audio.playRemove();
      }
      updateSelectedChipIndicator();
      refreshDiscRingsAndSlots();
    }
    window.handleCenterCoreClick = handleCenterCoreClick;

    function handleCenterHubMouseDown(discKey, e) {
      if (e.button !== 0) return;
      const state = CENTER_CORE_STATE[discKey];
      if (!state.charged && !state.placedAtDock) return;

      const startX = e.clientX;
      const startY = e.clientY;

      const onCheckMove = (me) => {
        if (Math.hypot(me.clientX - startX, me.clientY - startY) > 5) {
          window.removeEventListener('mousemove', onCheckMove);
          window.removeEventListener('mouseup', onCancelCheck);
          if (state.placedAtDock && OUTER_DOCK_SLOTS[state.placedAtDock]) {
            OUTER_DOCK_SLOTS[state.placedAtDock].coreChip = null;
            state.placedAtDock = null;
          }
          state.detached = true;
          refreshDiscRingsAndSlots();
          startCorePointerDrag(discKey, me);
        }
      };
      const onCancelCheck = () => {
        window.removeEventListener('mousemove', onCheckMove);
        window.removeEventListener('mouseup', onCancelCheck);
      };
      window.addEventListener('mousemove', onCheckMove);
      window.addEventListener('mouseup', onCancelCheck);
    }
    window.handleCenterHubMouseDown = handleCenterHubMouseDown;

    function startCorePointerDrag(discKey, startEvent) {
      if (startEvent) {
        startEvent.preventDefault();
        startEvent.stopPropagation();
      }
      const dMeta = DISC_CONFIG[discKey];
      const coreChipId = 'core_' + discKey;
      const spec = CHIP_SPECS[coreChipId];
      const stage = document.getElementById('diskStage');
      const svgBeams = document.getElementById('svgBeamsGroup');
      if (!stage || !spec) return;

      IS_DRAGGING_CORE = true;
      DRAGGING_CORE_DISC = discKey;
      CENTER_CORE_STATE[discKey].detached = true;
      ACTIVE_SELECTED_CHIP = coreChipId;
      updateSelectedChipIndicator();

      for (const [dockKey, dSlot] of Object.entries(OUTER_DOCK_SLOTS)) {
        if (dSlot.discKey !== discKey) {
          const el = document.getElementById('dockEl_' + dockKey);
          if (el) el.classList.add('ring-4', 'ring-sky-400', 'scale-125', 'z-50');
        }
      }

      let ghost = document.getElementById('coreDragGhostEl');
      if (ghost) ghost.remove();
      ghost = document.createElement('div');
      ghost.id = 'coreDragGhostEl';
      ghost.className = 'fixed pointer-events-none z-[9999] -translate-x-1/2 -translate-y-1/2 flex flex-col items-center';
      ghost.innerHTML = `
        <div class="w-11 h-11 rounded-full ${spec.matClass} flex items-center justify-center shadow-2xl scale-115 animate-pulse">
          <i class="fa-solid ${spec.icon} text-sm"></i>
        </div>
        <span class="mt-1 px-2 py-0.5 rounded-full bg-slate-900/95 text-amber-300 border border-sky-400 text-[10px] font-mono-sci font-bold shadow-lg whitespace-nowrap">
          ⚡ 拖至其他盘【独立外圈插槽】松开即连上！
        </span>
      `;
      document.body.appendChild(ghost);

      // 动态预览导入线：直接从源盘中心拉向鼠标/目标盘独立外圈插槽！
      let previewLine = document.getElementById('coreDragPreviewBeam');
      if (previewLine) previewLine.remove();
      previewLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      previewLine.setAttribute('id', 'coreDragPreviewBeam');
      previewLine.setAttribute('stroke', '#00f0ff');
      previewLine.setAttribute('stroke-width', '3.5');
      previewLine.setAttribute('stroke-dasharray', '8 5');
      previewLine.setAttribute('opacity', '0.9');
      if (svgBeams) svgBeams.appendChild(previewLine);

      previewLine.setAttribute('x1', dMeta.cx);
      previewLine.setAttribute('y1', dMeta.cy);

      let hoveredTargetDockKey = null;

      function updateDragPos(clientX, clientY) {
        ghost.style.left = `${clientX}px`;
        ghost.style.top = `${clientY}px`;

        const rect = stage.getBoundingClientRect();
        const scale = rect.width / STAGE_LOGICAL_W;
        const stageX = (clientX - rect.left) / scale;
        const stageY = (clientY - rect.top) / scale;

        let bestDockKey = null;
        let minDist = 75;
        for (const [dockKey, dSlot] of Object.entries(OUTER_DOCK_SLOTS)) {
          if (dSlot.discKey === discKey) continue;
          const pos = getOuterDockWorldPos(dockKey);
          const d = Math.hypot(stageX - pos.x, stageY - pos.y);
          if (d < minDist) {
            minDist = d;
            bestDockKey = dockKey;
          }
        }

        if (hoveredTargetDockKey && hoveredTargetDockKey !== bestDockKey) {
          const prevEl = document.getElementById('dockEl_' + hoveredTargetDockKey);
          if (prevEl) prevEl.classList.remove('scale-150', 'ring-amber-400');
        }
        hoveredTargetDockKey = bestDockKey;

        if (bestDockKey) {
          const snapPos = getOuterDockWorldPos(bestDockKey);
          previewLine.setAttribute('x2', snapPos.x);
          previewLine.setAttribute('y2', snapPos.y);
          previewLine.setAttribute('stroke', '#fbbf24');
          const curEl = document.getElementById('dockEl_' + bestDockKey);
          if (curEl) curEl.classList.add('scale-150', 'ring-amber-400');
        } else {
          previewLine.setAttribute('x2', stageX);
          previewLine.setAttribute('y2', stageY);
          previewLine.setAttribute('stroke', '#00f0ff');
        }
      }

      updateDragPos(startEvent.clientX, startEvent.clientY);

      const onMouseMove = (me) => {
        updateDragPos(me.clientX, me.clientY);
      };

      const onMouseUp = () => {
        window.removeEventListener('mousemove', onMouseMove);
        window.removeEventListener('mouseup', onMouseUp);
        IS_DRAGGING_CORE = false;
        DRAGGING_CORE_DISC = null;
        LAST_CORE_DRAG_TIMESTAMP = Date.now();

        if (ghost) ghost.remove();
        if (previewLine) previewLine.remove();

        for (const dockKey of Object.keys(OUTER_DOCK_SLOTS)) {
          const el = document.getElementById('dockEl_' + dockKey);
          if (el) el.classList.remove('ring-4', 'ring-sky-400', 'ring-amber-400', 'scale-125', 'scale-150', 'z-50');
        }

        if (hoveredTargetDockKey) {
          placeCoreInOuterDock(hoveredTargetDockKey, coreChipId);
        } else {
          // 未放入其他盘外圈槽时，自动归位回本盘中心！
          CENTER_CORE_STATE[discKey].placedAtDock = null;
          CENTER_CORE_STATE[discKey].detached = false;
          ACTIVE_SELECTED_CHIP = null;
          updateSelectedChipIndicator();
          refreshDiscRingsAndSlots();
        }
      };

      window.addEventListener('mousemove', onMouseMove);
      window.addEventListener('mouseup', onMouseUp);
    }
    window.startCorePointerDrag = startCorePointerDrag;

      const detachedLayer = document.getElementById('detachedCoresLayer');

      for (const [dKey, dMeta] of Object.entries(DISC_CONFIG)) {

        }
      }
    }
